def build(ctx):
    ctx.length(36)

    DROP = 0.009  # hip dip at contact (m); stance legs bend to keep feet on the ground

    def spine(f, h, side, twist):
        ctx.key(f, 'Spine02', turn=[('right', h * 0.4), ('forward', side * 0.5)])
        ctx.key(f, 'Spine01', turn=[('right', h * 0.35), ('forward', side * 0.5), ('up', twist)])
        ctx.key(f, 'Spine', turn=[('right', h * 0.25)])
        ctx.key(f, 'neck', turn=[('right', -h * 0.5), ('forward', -side * 0.6)])

    def staff(f, h, tilt, lift):
        # tilt = staff bottom forward (deg, world); lift raises the hand so the tip never enters the ground
        ctx.key(f, 'RightShoulder', turn=[('forward', -10 * lift)])
        ctx.key(f, 'RightArm', turn=[('right', tilt - h)])
        ctx.key(f, 'RightForeArm', turn=[('forward', -12 * lift)])
        ctx.key(f, 'RightHand', turn=[('forward', 22 * lift)])

    def leg(f, side, thigh, shin, foot, interp='BEZIER'):
        ctx.key(f, side + 'UpLeg', turn=[('right', thigh)], interp=interp)
        ctx.key(f, side + 'Leg', turn=[('right', shin)], interp=interp)
        ctx.key(f, side + 'Foot', turn=[('right', foot)], interp=interp)

    def stance(f, side, k, interp='BEZIER'):
        # knee bend that exactly absorbs a hip dip of k*DROP with the foot staying put and flat
        if side == 'Left':
            leg(f, 'Left', 4.5 * k, -8.3 * k, 3.8 * k, interp)
        else:
            leg(f, 'Right', 6.0 * k, -12.7 * k, 6.7 * k, interp)

    # frame: hunch, sway(+ = toward right), chest twist, staff tilt, staff lift, hip dip
    beats = [
        (3, -4, 3, 1, 2, 0.4, 0.0),
        (8, -7, 5, 5, 12, 1.0, 0.0),     # passing: staff swung forward in the air with the left foot
        (11, -8, 3, 5, 10, 0.6, 0.0),
        (13, -12, 0, 4, 8, 0.25, 1.0),   # CONTACT: left foot + staff plant together, sag onto stick
        (16, -9, -1, 3, 7, 0.25, 0.5),
        (21, -6, -4, 0, 5, 0.25, 0.15),
        (26, -6, -5, -3, 3, 0.25, 0.0),  # passing: right foot up, leaning on stick + left leg
        (29, -6, -3, -3, 3, 0.25, 0.0),
        (31, -10, 0, -2, 2, 0.25, 1.0),  # CONTACT: right foot thud
        (34, -5, 0, -1, 1, 0.15, 0.5),
    ]
    for f, h, side, twist, tilt, lift, k in beats:
        spine(f, h, side, twist)
        staff(f, h, tilt, lift)
        ctx.key(f, 'Hips', move=(0, -DROP * k, 0))

    # left leg: steps with the staff (frames 3-13), then stance
    leg(3, 'Left', 8, -10, 2)
    leg(8, 'Left', 25, -35, 5)
    leg(11, 'Left', 12, -16, 4, 'LINEAR')
    stance(13, 'Left', 1.0)
    stance(16, 'Left', 0.5)
    stance(21, 'Left', 0.15)
    stance(26, 'Left', 0.0)
    stance(29, 'Left', 0.0)
    stance(31, 'Left', 1.0)
    stance(34, 'Left', 0.5)

    # right leg: stance, then a quicker, stiffer step (frames 21-31)
    stance(3, 'Right', 0.0)
    stance(8, 'Right', 0.0)
    stance(11, 'Right', 0.0)
    stance(13, 'Right', 1.0)
    stance(16, 'Right', 0.5)
    leg(21, 'Right', 10, -16, 6)
    leg(26, 'Right', 35, -55, 15)
    leg(29, 'Right', 14, -22, 8, 'LINEAR')
    stance(31, 'Right', 1.0)
    stance(34, 'Right', 0.5)

    # head: eye on the stick as it plants, nod on each thud
    for f, t in ((3, [('right', 0)]),
                 (8, [('up', -8), ('right', -4)]),
                 (13, [('up', -6), ('right', -10)]),
                 (16, [('up', -2), ('right', 3)]),
                 (26, [('up', 4), ('right', 0)]),
                 (31, [('up', 2), ('right', -8)]),
                 (34, [('right', 2)])):
        ctx.key(f, 'Head', turn=t)

    # free left arm flaps out for balance on each contact, grumpy wobble
    for f, out, fwd, bend in ((3, 8, -5, 15), (8, 12, -12, 25), (13, 28, 8, 45), (16, 22, 5, 40),
                              (21, 12, 0, 25), (26, 10, 10, 30), (31, 24, 6, 45), (34, 14, 3, 25)):
        ctx.key(f, 'LeftArm', turn=[('forward', out), ('right', fwd)])
        ctx.key(f, 'LeftForeArm', turn=[('right', bend)])
    for f, a in ((11, 10), (13, 15), (16, -20), (29, 10), (31, 12), (34, -15)):
        ctx.key(f, 'LeftHand', turn=[('right', a)])

    ctx.note('Walk: in-place hobble. Staff lifts f3, swings forward f8, plants with left foot f13 and drifts back '
             'while planted; right foot steps f21-31. Hip dips 9mm at contacts with matching knee bends.')
