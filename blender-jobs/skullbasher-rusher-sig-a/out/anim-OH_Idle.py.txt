import math
from mathutils import Vector, Quaternion

AX = {'right': (1, 0, 0), 'left': (-1, 0, 0), 'up': (0, 1, 0),
      'down': (0, -1, 0), 'forward': (0, 0, 1), 'back': (0, 0, -1)}
Z = [('up', 0)]


def tq(turns):
    # (right, up, forward) is left-handed, so a physical right-hand-rule turn is a negative math angle
    q = Quaternion()
    for a, d in turns:
        q = Quaternion(Vector(AX[a]), math.radians(-d)) @ q
    return q


def solve(ctx, f, hm, items, interp):
    W = {}

    def world(b):
        if not b:
            return Quaternion()
        if b in W:
            return W[b]
        return world(ctx.parent(b))

    ctx.key(f, 'Hips', move=hm, interp=interp)
    for b, kind, val in items:
        P = world(ctx.parent(b))
        rest = Vector(ctx.bones[b]['points'])
        if kind == 'turn':
            L = tq(val)
            ctx.key(f, b, turn=list(val), interp=interp)
        elif kind == 'aim':
            la = (P.inverted() @ Vector(val)).normalized()
            L = rest.rotation_difference(la)
            ctx.key(f, b, aim=(la.x, la.y, la.z), interp=interp)
        else:
            L = P.inverted() @ val
            e = L.to_euler('XYZ')
            ctx.key(f, b, turn=[('right', -math.degrees(e.x)), ('up', -math.degrees(e.y)),
                                ('forward', -math.degrees(e.z))], interp=interp)
        W[b] = P @ L


def leg_items(ctx, side, hm, target, rot):
    B = ctx.bones
    th, sh, ft = side + 'UpLeg', side + 'Leg', side + 'Foot'
    H0 = Vector(B[th]['head']); K0 = Vector(B[sh]['head']); A0 = Vector(B[ft]['head'])
    L1 = (K0 - H0).length; L2 = (A0 - K0).length
    r = (A0 - H0).normalized()
    pole = (K0 - H0) - r * (K0 - H0).dot(r)
    H = H0 + Vector(hm); T = Vector(target)
    d = T - H
    dist = max(min(d.length, (L1 + L2) * 0.995), abs(L1 - L2) + 0.01)
    dn = d.normalized()
    pole = pole - dn * pole.dot(dn)
    pole.normalize()
    ca = max(-1.0, min(1.0, (L1 * L1 + dist * dist - L2 * L2) / (2 * L1 * dist)))
    sa = math.sqrt(1 - ca * ca)
    K = H + dn * (L1 * ca) + pole * (L1 * sa)
    E = H + dn * dist
    return [(th, 'aim', tuple(K - H)), (sh, 'aim', tuple(E - K)), (ft, 'world', rot)]


def pose(ctx, f, p, interp='BEZIER'):
    B = ctx.bones
    hm = p.get('hips', (0, 0, 0))
    items = [('Spine02', 'turn', p.get('s2', Z)), ('Spine01', 'turn', p.get('s1', Z)),
             ('Spine', 'turn', p.get('s0', Z)), ('neck', 'turn', p.get('neck', Z)),
             ('Head', 'turn', p.get('head', Z)),
             ('LeftShoulder', 'turn', p.get('lsh', Z)),
             ('LeftArm', 'aim', p['lu']), ('LeftForeArm', 'aim', p['lf']),
             ('LeftHand', 'aim', p.get('lh', p['lf'])),
             ('RightShoulder', 'turn', p.get('rsh', Z)),
             ('RightArm', 'aim', p['ru']), ('RightForeArm', 'aim', p['rf']),
             ('RightHand', 'aim', p.get('rh', p['rf']))]
    items += leg_items(ctx, 'Left', hm, p.get('lfoot', tuple(B['LeftFoot']['head'])), tq(p.get('lrot', Z)))
    items += leg_items(ctx, 'Right', hm, p.get('rfoot', tuple(B['RightFoot']['head'])), tq(p.get('rrot', Z)))
    solve(ctx, f, hm, items, interp)
