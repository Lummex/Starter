def build(ctx):
    ctx.length(60)
    hips = ctx.bone('hips')
    sp_lo = 'Spine02'
    chest = ctx.bone('spine')
    head = ctx.bone('head')
    neck = ctx.bone('neck')
    r_up = ctx.bone('right upperarm'); r_fore = ctx.bone('right forearm'); r_hand = ctx.bone('right hand')
    l_fore = ctx.bone('left forearm'); l_hand = ctx.bone('left hand')

    # weight shift: onto right foot, then left foot, then settle (small, feet stay put)
    ctx.key(8, hips, move=(0.01, -0.004, 0))
    ctx.key(22, hips, move=(0.01, -0.004, 0))
    ctx.key(30, hips, move=(-0.01, -0.004, 0))
    ctx.key(44, hips, move=(-0.01, -0.004, 0))
    ctx.key(52, hips, move=(0, -0.006, 0))

    # sneaky hunch in the middle of the loop
    ctx.key(6, sp_lo, turn=[('right', -8)])
    ctx.key(50, sp_lo, turn=[('right', -8)])
    ctx.key(8, chest, turn=[('forward', -3), ('right', -6)])
    ctx.key(22, chest, turn=[('forward', -3), ('right', -6)])
    ctx.key(30, chest, turn=[('forward', 3), ('right', -6), ('up', -6)])
    ctx.key(44, chest, turn=[('forward', 3), ('right', -6), ('up', -6)])

    # head: snap left, sniff, snap right, look at dagger, sniff at camera
    H = [
        (3, [('up', 38), ('right', -5)]),
        (10, [('up', 35), ('right', -5)]),
        (12, [('up', 32), ('right', -14)]),
        (14, [('up', 32), ('right', 2)]),
        (16, [('up', 32), ('right', -14)]),
        (18, [('up', 32), ('right', 0)]),
        (21, [('up', -42), ('forward', 8)]),
        (28, [('up', -38), ('forward', 6)]),
        (31, [('up', -22), ('right', -8)]),
        (42, [('up', -25), ('right', -6)]),
        (45, [('up', 2), ('right', -10)]),
        (47, [('up', 0), ('right', -18)]),
        (49, [('up', 0), ('right', -5)]),
        (51, [('up', 0), ('right', -18)]),
        (53, [('up', 0), ('right', -5)]),
        (57, [('up', 4), ('right', 2)]),
    ]
    for f, t in H:
        ctx.key(f, head, turn=t)
    # neck adds a little forward poke on the sniffs
    ctx.key(10, neck, turn=[('right', -6)])
    ctx.key(19, neck, turn=[('right', 0)])
    ctx.key(45, neck, turn=[('right', -8)])
    ctx.key(55, neck, turn=[('right', 0)])

    # left paw curls up to the chest while sniffing (rat paws)
    ctx.key(7, l_fore, turn=[('right', 40)])
    ctx.key(20, l_fore, turn=[('right', 40)])
    ctx.key(26, l_fore, turn=[('right', 0)])
    ctx.key(10, l_hand, turn=[('right', 25)])
    ctx.key(13, l_hand, turn=[('right', 5)])
    ctx.key(16, l_hand, turn=[('right', 25)])
    ctx.key(24, l_hand, turn=[('right', 0)])

    # dagger flick: anticipate, raise, flick out twice with a spin, settle
    ctx.key(28, r_up, turn=[('right', 0)])
    ctx.key(32, r_up, turn=[('right', 24), ('forward', -16)])
    ctx.key(42, r_up, turn=[('right', 22), ('forward', -14)])
    ctx.key(48, r_up, turn=[('right', 0)])
    ctx.key(28, r_fore, turn=[('right', 0)])
    ctx.key(30, r_fore, turn=[('right', -8)])
    ctx.key(32, r_fore, turn=[('right', 62)])
    ctx.key(40, r_fore, turn=[('right', 56)])
    ctx.key(46, r_fore, turn=[('right', 0)])
    ctx.key(31, r_hand, turn=[('forward', 0)])
    ctx.key(33, r_hand, turn=[('forward', 10)])
    ctx.key(35, r_hand, turn=[('forward', -40)], twist=70)
    ctx.key(37, r_hand, turn=[('forward', -28)], twist=70)
    ctx.key(39, r_hand, turn=[('forward', -45)], twist=0)
    ctx.key(41, r_hand, turn=[('forward', -22)], twist=0)
    ctx.key(46, r_hand, turn=[('forward', 0)])
    ctx.note('Idle: snap-look left + sniffs (3-18), snap right (21), dagger flick/spin (30-41), sniff at camera (45-53).')
