def build(ctx):
    ctx.length(90)
    H = round(0.12 * ctx.height, 3)
    hips = ctx.bone('hips'); spine = ctx.bone('spine'); neck = ctx.bone('neck'); head = ctx.bone('head')
    lu = ctx.bone('left upperarm'); ru = ctx.bone('right upperarm')
    lf = ctx.bone('left forearm'); rf = ctx.bone('right forearm')
    lh = ctx.bone('left hand'); rh = ctx.bone('right hand')
    lt = ctx.bone('left thigh'); rt = ctx.bone('right thigh')
    ls = ctx.bone('left shin'); rs = ctx.bone('right shin')
    lft = ctx.bone('left foot'); rft = ctx.bone('right foot')

    def pose(f, lean, dy, arm, out, bend, thigh, shin, foot, look):
        # lean on the hips tilts the whole body; negative = forward
        ctx.key(f, hips, turn=[('right', lean)], move=(0, H + dy, 0))
        ctx.key(f, spine, turn=[('right', lean * 0.12)])
        # head counters the lean so she keeps looking ahead
        ctx.key(f, neck, turn=[('right', look * 0.6)])
        ctx.key(f, head, turn=[('right', look * 0.4)])
        ctx.key(f, lu, turn=[('forward', out), ('right', arm)])
        ctx.key(f, ru, turn=[('forward', -out), ('right', arm)])
        ctx.key(f, lf, turn=[('right', bend)])
        ctx.key(f, rf, turn=[('right', bend)])
        ctx.key(f, lh, turn=[('right', min(0, arm) * 0.3)])
        ctx.key(f, rh, turn=[('right', min(0, arm) * 0.3)])
        for t, s, ft in ((lt, ls, lft), (rt, rs, rft)):
            ctx.key(f, t, turn=[('right', thigh)])
            ctx.key(f, s, turn=[('right', shin)])
            ctx.key(f, ft, turn=[('right', foot)])

    #        f   lean   dy     arm  out bend thigh shin foot look
    pose(0,   0,  0.0,     0,  10, 12,  10, -22, -30,  0)    # hover
    pose(9,   5, -0.012,  15,  12, 20,  16, -26, -30, -3)    # anticipation: sit back, arms forward
    pose(18, -25, 0.006, -58,   6,  4, -12, -32, -45, 18)    # push into the lean (overshoot)
    # cruise: wing-beat bob, 3 beats; down-stroke sinks, leans a hair more, arms press back
    for f, down in ((24, 0), (29, 1), (34, 0), (39, 1), (44, 0), (49, 1), (54, 0), (59, 1), (64, 0)):
        if down:
            pose(f, -23, -0.016, -56, 8, 6, -10, -34, -42, 16)
        else:
            pose(f, -21,  0.004, -50, 6, 8, -8, -28, -40, 14)
    pose(73,  6, -0.008,  12,  12, 18,  18, -24, -30, -4)    # brake: tip back upright, legs swing under
    pose(81, -2,  0.003,   2,  10, 13,  11, -22, -30,  1)    # settle
    pose(90,  0,  0.0,     0,  10, 12,  10, -22, -30,  0)    # hover (loops)
    ctx.note('Fly: hover f0, anticipation f9, lean -25 f18, cruise -21..-23 with 3 wing-beat bobs f24-64, brake f73, hover f90')
