def build(ctx):
    ctx.length(48)

    hips = ctx.bone('hips')
    spines = [ctx.bone('Spine02'), ctx.bone('Spine01'), ctx.bone('spine')]
    neck = ctx.bone('neck')
    head = ctx.bone('head')
    r_sh = ctx.bone('right shoulder')
    r_up = ctx.bone('right upperarm')
    r_fo = ctx.bone('right forearm')
    r_ha = ctx.bone('right hand')
    l_up = ctx.bone('left upperarm')
    l_fo = ctx.bone('left forearm')
    l_th = ctx.bone('left thigh')
    l_sn = ctx.bone('left shin')
    l_ft = ctx.bone('left foot')
    r_th = ctx.bone('right thigh')
    r_sn = ctx.bone('right shin')
    r_ft = ctx.bone('right foot')

    RELEASE = 19

    # Preview skull in the right hand, vanishes on release.
    if 'skull-mace' in ctx.parts:
        ctx.attach('skull-mace', r_ha, length=0.12, hide_at=RELEASE)

    def torso(f, twist, bend, side=0.0, interp='BEZIER'):
        # spread the twist/bend over the three spine bones; head counters part of the twist
        for s in spines:
            ctx.key(f, s, turn=[('up', twist / 3.0), ('right', bend / 3.0), ('forward', side / 3.0)], interp=interp)
        ctx.key(f, neck, turn=[('up', -twist * 0.2)], interp=interp)
        ctx.key(f, head, turn=[('up', -twist * 0.3), ('right', -bend * 0.4)], interp=interp)

    def legs(f, crouch, interp='BEZIER'):
        # knee bend that keeps the feet roughly under the body when the hips drop
        for th, sn, ft in ((l_th, l_sn, l_ft), (r_th, r_sn, r_ft)):
            ctx.key(f, th, turn=[('right', crouch)], interp=interp)
            ctx.key(f, sn, turn=[('right', -2.0 * crouch)], interp=interp)
            ctx.key(f, ft, turn=[('right', crouch)], interp=interp)

    def pose(f, hip_move, crouch, twist, bend, side, up_aim, fo_aim, hand_turn,
             l_up_aim, l_fo_aim, sh_turn=0.0, interp='BEZIER'):
        ctx.key(f, hips, move=hip_move, interp=interp)
        legs(f, crouch, interp)
        torso(f, twist, bend, side, interp)
        ctx.key(f, r_sh, turn=[('forward', sh_turn)], interp=interp)
        ctx.key(f, r_up, aim=up_aim, interp=interp)
        ctx.key(f, r_fo, aim=fo_aim, interp=interp)
        ctx.key(f, r_ha, turn=[('right', hand_turn)], interp=interp)
        ctx.key(f, l_up, aim=l_up_aim, interp=interp)
        ctx.key(f, l_fo, aim=l_fo_aim, interp=interp)

    # 5: anticipation - dip, start twisting away, skull lifts
    pose(5, (0.008, -0.018, -0.006), 14, -25, 6, 0,
         (0.55, 0.15, -0.55), (0.3, 0.6, -0.6), 15,
         (-0.6, -0.2, 0.6), (-0.3, 0.1, 0.95))

    # 11: full wind-up - big twist away, lean back, elbow high, skull cocked far behind the head
    pose(11, (0.015, -0.025, -0.015), 20, -66, 16, 8,
         (0.7, 0.68, -0.25), (-0.15, 0.2, -0.97), 35,
         (-0.45, 0.35, 0.82), (-0.2, 0.45, 0.87), sh_turn=12)
    # 14: hold the coil (tiny extra stretch so it does not die)
    pose(14, (0.016, -0.026, -0.017), 21, -70, 18, 9,
         (0.68, 0.7, -0.28), (-0.2, 0.18, -0.96), 38,
         (-0.45, 0.38, 0.8), (-0.2, 0.48, 0.85), sh_turn=14)

    # 17: whip - hips drive forward/left, chest unwinds, elbow leads over the shoulder, forearm drags behind
    pose(17, (-0.008, -0.02, 0.012), 16, 5, -6, -4,
         (0.5, 0.85, 0.15), (0.15, 0.55, -0.82), 30,
         (-0.6, -0.3, 0.3), (-0.4, -0.5, 0.4), sh_turn=10, interp='LINEAR')

    # 19: release at the front - arm snapped out forward/up, wrist flicks
    pose(RELEASE, (-0.016, -0.02, 0.02), 15, 32, -20, -6,
         (0.25, 0.4, 0.88), (0.1, 0.3, 0.95), -30,
         (-0.6, -0.55, -0.2), (-0.3, -0.7, -0.4), sh_turn=0)
    ctx.event(RELEASE, 'projectile_release', bone=r_ha)

    # 23: follow-through across the body, weight over the front (left) foot
    pose(23, (-0.02, -0.026, 0.024), 20, 48, -32, -8,
         (-0.25, -0.55, 0.8), (-0.6, -0.65, 0.45), -20,
         (-0.5, -0.65, -0.45), (-0.2, -0.8, -0.5))
    # 27: hold the follow-through
    pose(27, (-0.019, -0.025, 0.022), 19, 44, -29, -7,
         (-0.22, -0.6, 0.77), (-0.55, -0.7, 0.42), -15,
         (-0.5, -0.68, -0.42), (-0.2, -0.82, -0.48))

    # 35: recover, slight overshoot back past idle
    pose(35, (0.004, 0.004, -0.004), -2, -8, 5, 1,
         (0.2, -0.8, -0.5), (0.35, -0.93, -0.05), 5,
         (-0.37, -0.92, -0.05), (-0.03, -0.8, 0.6))

    # 42: settle toward idle (frame 48 returns to the base pose automatically)
    pose(42, (0.0, -0.002, 0.0), 1, 3, -1, 0,
         (0.14, -0.76, -0.63), (0.38, -0.92, 0.0), 0,
         (-0.37, -0.92, -0.1), (-0.03, -0.79, 0.61))

    ctx.note('Skull_Throw take 3: wind-up peak f11-14, whip f17, release f19, follow-through f23-27, settle f35-48.')
