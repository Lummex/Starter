import math


def build(ctx):
    ctx.length(120)

    hips = ctx.bone('hips')
    sp_lo = 'Spine02'
    sp_mid = 'Spine01'
    spine = ctx.bone('spine')
    neck = ctx.bone('neck')
    head = ctx.bone('head')
    r_up = ctx.bone('right upperarm'); r_fore = ctx.bone('right forearm'); r_hand = ctx.bone('right hand')
    l_up = ctx.bone('left upperarm'); l_fore = ctx.bone('left forearm'); l_hand = ctx.bone('left hand')
    r_thigh = ctx.bone('right thigh'); r_shin = ctx.bone('right shin'); r_foot = ctx.bone('right foot')
    l_thigh = ctx.bone('left thigh'); l_shin = ctx.bone('left shin'); l_foot = ctx.bone('left foot')

    leg_len = 0.322  # thigh + shin, for planted knee bends

    def planted_legs(f, bend, sway):
        # equal and opposite thigh / shin / foot turns keep the feet on their spot
        for th, sh, ft in ((r_thigh, r_shin, r_foot), (l_thigh, l_shin, l_foot)):
            ctx.key(f, th, turn=[('forward', -sway), ('right', bend)])
            ctx.key(f, sh, turn=[('right', -2 * bend)])
            ctx.key(f, ft, turn=[('right', bend)])

    def drop(bend):
        return -leg_len * (1 - math.cos(math.radians(bend)))

    # ---------- part 1: dizzy wobble, feet planted (fall-over f01-f06) ----------
    wobble = [
        # frame, hip sway, knee bend, spine side, spine fwd, neck side, head side, head fwd
        (8, 4, 4, -6, -2, 8, 6, -2),
        (16, -5, 6, 8, -4, -10, -8, -6),
        (24, 5, 9, -8, -6, 12, 8, -10),
        (32, -4, 13, 6, -8, -8, -10, -8),
    ]
    for f, sway, bend, s_side, s_fwd, n_side, h_side, h_fwd in wobble:
        ctx.key(f, hips, turn=[('forward', sway)], move=(0, drop(bend), 0))
        ctx.key(f, sp_lo, turn=[('forward', s_side * 0.5), ('right', s_fwd * 0.5)])
        ctx.key(f, spine, turn=[('forward', s_side), ('right', s_fwd)])
        ctx.key(f, neck, turn=[('forward', n_side), ('right', h_fwd * 0.5)])
        ctx.key(f, head, turn=[('forward', h_side), ('right', h_fwd)])
        planted_legs(f, bend, sway)
        swing = sway * 2
        ctx.key(f, r_up, turn=[('forward', 4 + swing * 0.5), ('right', 4)])
        ctx.key(f, l_up, turn=[('forward', -4 + swing * 0.5), ('right', 4)])
        ctx.key(f, r_fore, turn=[('right', 6)])
        ctx.key(f, l_fore, turn=[('right', 8)])

    # ---------- knees buckle (f07-f08), feet still planted ----------
    for f, bend, s_fwd in ((40, 30, -12), (47, 46, -16)):
        ctx.key(f, hips, turn=[('right', 4)], move=(0, drop(bend), 0))
        ctx.key(f, sp_lo, turn=[('right', s_fwd * 0.4)])
        ctx.key(f, spine, turn=[('right', s_fwd)])
        ctx.key(f, neck, turn=[('forward', 6), ('right', -6)])
        ctx.key(f, head, turn=[('forward', 8), ('right', -10)])
        planted_legs(f, bend, 0)
        # board arm starts lifting so the board clears the floor when he sits
        ctx.key(f, r_up, aim=(0.75, -0.6, 0.2))
        ctx.key(f, r_fore, aim=(0.15, -0.92, 0.35))
        ctx.key(f, r_hand, turn=[('right', 0)])
        ctx.key(f, l_up, aim=(-0.55, -0.8, 0.15))
        ctx.key(f, l_fore, aim=(-0.3, -0.85, 0.35))
        ctx.key(f, l_hand, turn=[('right', 0)])

    # ---------- plop onto his bottom (f09-f10) ----------
    sit_thigh_r = (0.15, 0.02, 1.0); sit_thigh_l = (-0.15, 0.02, 1.0)
    sit_shin_r = (0.1, -0.4, 0.92); sit_shin_l = (-0.1, -0.4, 0.92)
    sit = [
        # frame, hip y, hip lean back, spine fwd, neck fwd
        (53, -0.300, -4, -14, -8),   # impact, squash forward
        (57, -0.296, 6, 0, -2),      # rebound
        (62, -0.298, 4, -4, -6),     # settle sitting (f10)
    ]
    for f, y, lean, s_fwd, n_fwd in sit:
        ctx.key(f, hips, turn=[('right', lean)], move=(0, y, -0.005))
        ctx.key(f, sp_lo, turn=[('right', s_fwd * 0.4)])
        ctx.key(f, spine, turn=[('right', s_fwd)])
        ctx.key(f, neck, turn=[('forward', 5), ('right', n_fwd)])
        ctx.key(f, head, turn=[('forward', 8), ('right', n_fwd)])
        ctx.key(f, r_thigh, aim=sit_thigh_r); ctx.key(f, l_thigh, aim=sit_thigh_l)
        ctx.key(f, r_shin, aim=sit_shin_r); ctx.key(f, l_shin, aim=sit_shin_l)
        ctx.key(f, r_foot, turn=[('right', 0)]); ctx.key(f, l_foot, turn=[('right', 0)])
        ctx.key(f, r_up, aim=(0.95, -0.25, 0.15))
        ctx.key(f, r_fore, aim=(0.1, -0.9, 0.4))
        ctx.key(f, r_hand, turn=[('right', 0)])
        ctx.key(f, l_up, aim=(-0.85, -0.5, 0.0))
        ctx.key(f, l_fore, aim=(-0.6, -0.8, 0.05))
        ctx.key(f, l_hand, turn=[('right', 10)])

    # ---------- tips back (f11-f12), legs come up a little ----------
    f = 68
    ctx.key(f, hips, turn=[('right', 32)], move=(0, -0.294, -0.015))
    ctx.key(f, sp_lo, turn=[('right', -2)])
    ctx.key(f, spine, turn=[('right', -6)])
    ctx.key(f, neck, turn=[('forward', 4), ('right', -10)])
    ctx.key(f, head, turn=[('forward', 6), ('right', -8)])
    ctx.key(f, r_thigh, aim=(0.15, -0.15, 1.0)); ctx.key(f, l_thigh, aim=(-0.15, -0.15, 1.0))
    ctx.key(f, r_shin, aim=(0.1, -0.5, 0.86)); ctx.key(f, l_shin, aim=(-0.1, -0.5, 0.86))
    ctx.key(f, r_foot, turn=[('right', 0)]); ctx.key(f, l_foot, turn=[('right', 0)])
    ctx.key(f, r_up, aim=(0.95, -0.2, 0.25))
    ctx.key(f, r_fore, aim=(0.2, -0.85, 0.5))
    ctx.key(f, r_hand, turn=[('right', 0)])
    ctx.key(f, l_up, aim=(-0.9, -0.35, 0.15))
    ctx.key(f, l_fore, aim=(-0.7, -0.6, 0.2))
    ctx.key(f, l_hand, turn=[('right', 10)])

    # lands on his back: overshoot, legs kick up, arms flop up
    f = 74
    ctx.key(f, hips, turn=[('right', 82)], move=(0, -0.288, -0.03))
    ctx.key(f, sp_lo, turn=[('right', -2)])
    ctx.key(f, spine, turn=[('right', -4)])
    ctx.key(f, neck, turn=[('right', -14)])
    ctx.key(f, head, turn=[('forward', 4), ('right', -8)])
    ctx.key(f, r_thigh, aim=(0.15, -0.75, 0.6)); ctx.key(f, l_thigh, aim=(-0.15, -0.75, 0.6))
    ctx.key(f, r_shin, aim=(0.12, -0.9, 0.3)); ctx.key(f, l_shin, aim=(-0.12, -0.9, 0.3))
    ctx.key(f, r_foot, turn=[('right', 0)]); ctx.key(f, l_foot, turn=[('right', 0)])
    ctx.key(f, r_up, aim=(0.9, -0.15, 0.2))
    ctx.key(f, r_fore, aim=(0.55, -0.7, 0.3))
    ctx.key(f, r_hand, turn=[('forward', 40), ('right', 8)])
    ctx.key(f, l_up, aim=(-0.85, -0.2, 0.3))
    ctx.key(f, l_fore, aim=(-0.75, -0.45, 0.25))
    ctx.key(f, l_hand, turn=[('right', 10)])

    # ---------- part 2: settle flat (lying-still f01-f04) ----------
    lying_thigh_r = (0.15, -0.97, 0.26); lying_thigh_l = (-0.15, -0.97, 0.26)
    lying_shin_r = (0.12, -0.99, -0.05); lying_shin_l = (-0.12, -0.99, -0.05)

    def lying(f, back, y, leg_lift, arm_lift):
        ctx.key(f, hips, turn=[('right', back)], move=(0, y, -0.03))
        ctx.key(f, sp_lo, turn=[('right', 2)])
        ctx.key(f, spine, turn=[('right', 2)])
        ctx.key(f, neck, turn=[('right', -8)])
        ctx.key(f, head, turn=[('forward', -6), ('right', -5)])
        ctx.key(f, r_thigh, aim=(lying_thigh_r[0], lying_thigh_r[1], lying_thigh_r[2] + leg_lift))
        ctx.key(f, l_thigh, aim=(lying_thigh_l[0], lying_thigh_l[1], lying_thigh_l[2] + leg_lift))
        ctx.key(f, r_shin, aim=lying_shin_r); ctx.key(f, l_shin, aim=lying_shin_l)
        ctx.key(f, r_foot, turn=[('forward', -8)]); ctx.key(f, l_foot, turn=[('forward', 8)])
        # board arm out to his right, board turned out flat beside him
        ctx.key(f, r_up, aim=(0.85, -0.45, -0.12 + arm_lift))
        ctx.key(f, r_fore, aim=(0.5, -0.85, 0.05 + arm_lift))
        ctx.key(f, r_hand, turn=[('forward', 60), ('right', 15)])
        # free arm flopped out to the side on the floor
        ctx.key(f, l_up, aim=(-0.8, -0.3, -0.45 + arm_lift))
        ctx.key(f, l_fore, aim=(-0.75, -0.55, -0.3 + arm_lift))
        ctx.key(f, l_hand, turn=[('right', 10)])

    lying(79, 74, -0.284, 0.12, 0.12)   # small bounce up off the floor
    lying(85, 79, -0.291, 0.0, -0.02)   # drops back, tiny overshoot
    lying(90, 78, -0.290, 0.0, 0.0)     # final pose
    lying(120, 78, -0.290, 0.0, 0.0)    # held completely still

    if 'frying-pan' in ctx.parts:
        ctx.attach('frying-pan', r_hand, length=0.35)
    ctx.note('Death: dizzy wobble 0-32, knees buckle 40-47 (feet planted), plop sit 53-62, '
             'tip back 68, land 74, bounce 79, settle 85-90, hold lying 90-120.')
